sealed class ProgressSink(Action<DistributionProgress> action) : IProgress<DistributionProgress> { public void Report(DistributionProgress value)=>action(value); }
sealed class Fake : HttpMessageHandler
{
    public Dictionary<string,byte[]> Map {get;} = new(); public List<string> Requests {get;} = new(); public bool Offline;
    protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage r,CancellationToken ct)
    {
        ct.ThrowIfCancellationRequested(); if(Offline) throw new HttpRequestException("fixture offline");
        Requests.Add(r.RequestUri!.AbsoluteUri);
        return Task.FromResult(new HttpResponseMessage(Map.ContainsKey(r.RequestUri.AbsoluteUri)?HttpStatusCode.OK:HttpStatusCode.NotFound) { RequestMessage=r, Content=new ByteArrayContent(Map.GetValueOrDefault(r.RequestUri.AbsoluteUri)??[]) });
    }
}
