[MethodImpl(MethodImplOptions.NoInlining)]
static bool Collected(WeakReference weak)
{
    for(var i=0;i<20 && weak.IsAlive;i++) { GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect(); Thread.Sleep(10); }
    return !weak.IsAlive;
}

[MethodImpl(MethodImplOptions.NoInlining)]
static string Run(ModuleLease lease) => lease.Module.Commands[0].ExecuteAsync(default).GetAwaiter().GetResult().Message;

[MethodImpl(MethodImplOptions.NoInlining)]
static Version VersionOf(ModuleLease lease) => lease.Module.Descriptor.Version;
